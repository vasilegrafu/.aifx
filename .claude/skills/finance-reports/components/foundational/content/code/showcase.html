def shape(rows):
    """Opening + movements must reach closing."""
    opening = rows[0]["cells"]
    closing = rows[-1]["cells"]
    for i, (o, c) in enumerate(zip(opening, closing)):
        moved = sum(r["cells"][i] for r in rows[1:-1])
        assert abs(o + moved - c) < 0.01, f"column {i} does not close"
